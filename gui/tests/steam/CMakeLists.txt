cmake_minimum_required(VERSION 3.16)
project(chiaki-steam-shortcut-tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTORCC ON)
find_package(Qt6 REQUIRED COMPONENTS Gui Qml)
set(repo "${CMAKE_CURRENT_LIST_DIR}/../../..")
include("${repo}/cmake/SteamTools.cmake")
# Compile the production command without the unrelated session and network backend.
file(READ "${repo}/gui/src/qmlbackend.cpp" backend)
string(FIND "${backend}" "void QmlBackend::createSteamShortcut(" begin)
string(FIND "${backend}" "\n#endif\n\nQString QmlBackend::openPsnLink" end)
math(EXPR length "${end} - ${begin}")
string(SUBSTRING "${backend}" ${begin} ${length} command)
configure_file(command.cpp.in command.cpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${repo}/gui/src/qmlbackend.cpp")
add_executable(chiaki-steam-shortcut-tests main.cpp "${CMAKE_CURRENT_BINARY_DIR}/command.cpp" "${repo}/gui/res/resources.qrc")
target_include_directories(chiaki-steam-shortcut-tests PRIVATE "${CMAKE_CURRENT_LIST_DIR}")
target_link_libraries(chiaki-steam-shortcut-tests PRIVATE cpp-steam-tools Qt6::Gui Qt6::Qml)
